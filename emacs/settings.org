*** Spelling

Flyspell in prog-mode checks comments and strings only (~flyspell-prog-mode~).
~flyspell-prog-mode~ uses ~syntax-ppss~ which does not work correctly with
tree-sitter modes, so we replace the predicate with a tree-sitter-aware one
that checks the node type at point.

#+begin_src emacs-lisp
  (defun ao/flyspell-tree-sitter-check-p ()
    "Return t only when point is inside a comment or string node.
  Used as `flyspell-generic-check-word-predicate' in tree-sitter modes
  where `syntax-ppss' does not reliably identify node types."
    (when-let* ((node (treesit-node-at (point)))
                (type (treesit-node-type node)))
      (string-match-p "comment\\|string" type)))

  (defun ao/flyspell-prog-mode-ts ()
    "Enable flyspell with a tree-sitter-aware predicate."
    (setq-local flyspell-generic-check-word-predicate
                #'ao/flyspell-tree-sitter-check-p)
    (flyspell-mode 1))

  (use-package flyspell
    :ensure nil
    :hook (python-ts-mode . ao/flyspell-prog-mode-ts))
#+end_src

*** Line numbers
#+begin_src emacs-lisp
  ;; Relative line numbers in prog-mode (useful with evil/avy jumps).
  ;; Disabled in text/org modes where line numbers add visual clutter.
  (defun ao/enable-relative-line-numbers ()
    ;; Skip *scratch* (a prog-mode buffer) so it stays clean: no numbers, no rule.
    (unless (string= (buffer-name) "*scratch*")
      (display-line-numbers-mode t)
      (setq-local display-line-numbers 'relative)
      ;; Continuous thin rule between the line-number gutter and the code, in the
      ;; theme's own border colour (`vertical-border').  A background-coloured 1px
      ;; stretch (not a character) so it spans the `line-spacing' gaps and reads as
      ;; one unbroken line.  Spaces pad each side.
      (let* ((color (face-attribute 'vertical-border :foreground nil t))
             (sep (concat " "
                          (propertize " " 'face (list :background color)
                                      'display '(space :width (1)))
